{{- /* Windowed pagination: first, last, and the pages around the current one, with … for the gaps. sm and up: 1 … 4 5 6 … 11 (current ±1) phones: 1 … 5 … 11 (current only, so it fits 320px with 44px tap targets) Prev/Next show only the arrow on phones; the text stays available to screen readers. */ -}} {{- $paginator := .Paginator -}} {{- if gt $paginator.TotalPages 1 -}} {{- $cur := $paginator.PageNumber -}} {{- $last := $paginator.TotalPages -}} {{- $btn := "inline-flex items-center justify-center gap-1 h-11 sm:h-10 text-sm rounded-xl transition-all" -}} {{- $link := "font-medium text-slate-700 bg-surface border border-slate-200 hover:bg-primary-50 hover:text-primary-700 hover:border-primary-200" -}} {{- $disabled := "font-medium text-slate-300 bg-slate-50 border border-slate-100 cursor-not-allowed" -}} {{- $edge := "w-11 sm:w-auto sm:px-4" -}} {{- $num := "w-11 sm:w-10" -}} {{- $ellipsis := "items-center justify-center w-6 h-11 sm:h-10 text-sm text-slate-500" -}}
{{- if $paginator.HasPrev }} Prev {{- else }} Prev {{- end }} {{- $prevWide := 0 -}} {{- $prevNarrow := 0 -}} {{- range $paginator.Pagers }} {{- $n := .PageNumber -}} {{- $edgePage := or (eq $n 1) (eq $n $last) -}} {{- $wide := or $edgePage (le (math.Abs (sub $n $cur)) 1) -}} {{- $narrow := or $edgePage (eq $n $cur) -}} {{- if $wide }} {{- /* Gap markers: a narrow-screen gap always exists when a wide one does. */ -}} {{- $gapWide := gt (sub $n $prevWide) 1 -}} {{- $gapNarrow := and $narrow (gt (sub $n $prevNarrow) 1) -}} {{- if and $gapWide $narrow }} … {{- else if $gapWide }} … {{- else if $gapNarrow }} … {{- end }} {{- $vis := cond $narrow "" "hidden sm:inline-flex" -}} {{- if eq $n $cur }} {{ $n }} {{- else }} {{ $n }} {{- end }} {{- $prevWide = $n -}} {{- if $narrow }}{{ $prevNarrow = $n }}{{ end -}} {{- end }} {{- end }} {{- if $paginator.HasNext }} Next {{- else }} Next {{- end }}
{{- end -}}
